// Gera a capa de compartilhamento (WhatsApp, Facebook, etc.) com as fontes e a foto da marca.
// O site usa a versão JPG salva em public/og-zenite.jpg (leve, ideal para o WhatsApp).
// Para refazer: copie este arquivo para src/app/og-capa/route.tsx, rode npm run dev, abra /og-capa,
// salve a imagem e converta para JPG leve (ex.: ffmpeg -i capa.png -q:v 4 public/og-zenite.jpg). Depois apague a rota.
import { ImageResponse } from "next/og";
import { readFile } from "node:fs/promises";
import { join } from "node:path";

const dir = (p: string) => join(process.cwd(), p);

export async function GET() {
  const [cinzel, hanken300, hanken400, mono, foto] = await Promise.all([
    readFile(dir("assets/fonts/Cinzel-600.woff")),
    readFile(dir("assets/fonts/Hanken-300.woff")),
    readFile(dir("assets/fonts/Hanken-400.woff")),
    readFile(dir("assets/fonts/PlexMono-400.woff")),
    readFile(dir("public/img/coronel-2.jpg")),
  ]);
  const fotoSrc = `data:image/jpeg;base64,${foto.toString("base64")}`;
  const marrom = "#3A271C";
  const creme = "#F4ECE2";
  const ouro = "#C29A52";

  return new ImageResponse(
    (
      <div style={{ width: "100%", height: "100%", display: "flex", background: marrom, padding: 28 }}>
        <div style={{ position: "relative", display: "flex", width: "100%", height: "100%", borderRadius: 26, overflow: "hidden" }}>
          {/* eslint-disable-next-line @next/next/no-img-element */}
          <img src={fotoSrc} alt="" width={1144} height={574} style={{ position: "absolute", inset: 0, width: "100%", height: "100%", objectFit: "cover" }} />
          <div style={{ position: "absolute", top: 0, left: 0, width: "100%", height: "100%", display: "flex", backgroundImage: "linear-gradient(90deg, rgba(22,15,10,0.93) 0%, rgba(22,15,10,0.82) 42%, rgba(22,15,10,0.35) 72%, rgba(22,15,10,0.05) 100%)" }} />
          <div style={{ position: "absolute", left: 0, bottom: 0, width: "100%", height: "45%", display: "flex", backgroundImage: "linear-gradient(0deg, rgba(22,15,10,0.7) 0%, rgba(22,15,10,0) 100%)" }} />

          <div style={{ position: "relative", display: "flex", flexDirection: "column", justifyContent: "space-between", width: "100%", height: "100%", padding: "52px 60px" }}>
            {/* logotipo */}
            <div style={{ display: "flex", flexDirection: "column", alignItems: "flex-start", color: creme }}>
              <div style={{ fontFamily: "Cinzel", fontSize: 44, letterSpacing: 13, lineHeight: 1 }}>ZÊNITE</div>
              <div style={{ display: "flex", alignItems: "center", marginTop: 14, gap: 12, marginLeft: 4 }}>
                <div style={{ width: 34, height: 2, background: ouro }} />
                <div style={{ fontFamily: "Cinzel", fontSize: 16, letterSpacing: 9, color: creme }}>IMÓVEIS</div>
                <div style={{ width: 34, height: 2, background: ouro }} />
              </div>
            </div>

            {/* frase */}
            <div style={{ display: "flex", flexDirection: "column", color: "#fff", maxWidth: 700 }}>
              <div style={{ fontFamily: "Hanken", fontWeight: 300, fontSize: 70, lineHeight: 0.98, letterSpacing: -2.6 }}>Imóveis em Itanhaém, com quem conhece o litoral.</div>
              <div style={{ fontFamily: "Hanken", fontWeight: 400, fontSize: 26, marginTop: 22, color: "rgba(255,255,255,0.88)" }}>
                Chácaras na praia, casas e financiamento bancário.
              </div>
            </div>

            {/* rodapé */}
            <div style={{ display: "flex", alignItems: "center", gap: 18, fontFamily: "Plex", fontSize: 17, letterSpacing: 2.5, color: creme }}>

              <div style={{ display: "flex", opacity: 0.85 }}>ELAINE NUNES · CRECI 314779-F</div>
            </div>
          </div>
        </div>
      </div>
    ),
    {
      width: 1200,
      height: 630,
      fonts: [
        { name: "Cinzel", data: cinzel, weight: 600, style: "normal" },
        { name: "Hanken", data: hanken300, weight: 300, style: "normal" },
        { name: "Hanken", data: hanken400, weight: 400, style: "normal" },
        { name: "Plex", data: mono, weight: 400, style: "normal" },
      ],
    },
  );
}
